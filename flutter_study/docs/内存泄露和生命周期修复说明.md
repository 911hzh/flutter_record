# 内存泄露和生命周期问题修复说明

## 📋 发现的问题

### 问题 1：FlutterWidgetVC.swift 的内存泄露风险

#### 🔴 原始问题：

1. **缺少 `setupEventChannel` 方法实现**
   - 第 110 行调用了不存在的方法，导致编译错误

2. **循环引用风险**
   ```swift
   channel?.setMethodCallHandler({ call, result in
       // ❌ 没有使用 [weak self]
   })
   ```

3. **重复调用 result**
   ```swift
   if method == "getLocationInfomation" {
       result(data)
   }
   result("setMethodCallHandler result")  // ❌ 会重复调用
   ```

4. **强制销毁导致生命周期异常**
   ```swift
   flutterVC.engine.destroyContext()  // ❌ 立即销毁，Flutter Widget 无法正常清理
   ```

#### ✅ 修复方案：

1. **添加 TimerStreamHandler 类**
   ```swift
   class TimerStreamHandler: NSObject, FlutterStreamHandler {
       private var eventSink: FlutterEventSink?
       private var timer: Timer?
       
       func onListen(withArguments arguments: Any?, 
                     eventSink: @escaping FlutterEventSink) -> FlutterError? {
           timer = Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] _ in
               // 使用 [weak self] 避免循环引用
               guard let self = self else { return }
               eventSink(data)
           }
           return nil
       }
       
       func onCancel(withArguments arguments: Any?) -> FlutterError? {
           timer?.invalidate()  // 正确清理
           return nil
       }
   }
   ```

2. **实现 setupEventChannel 方法**
   ```swift
   private func setupEventChannel(controller: FlutterViewController) {
       let eventChannel = FlutterEventChannel(
           name: "FlutterNativeEventChannelBridge",
           binaryMessenger: controller.binaryMessenger
       )
       eventChannel.setStreamHandler(TimerStreamHandler())
   }
   ```

3. **修复 MethodChannel 的循环引用**
   ```swift
   channel?.setMethodCallHandler({ [weak self] call, result in
       guard self != nil else {
           result(FlutterMethodNotImplemented)
           return
       }
       
       if method == "getLocationInfomation" {
           result(data)
           return  // ✅ 避免重复调用
       }
       
       result("setMethodCallHandler result")
   })
   ```

4. **延迟销毁 FlutterEngine**
   ```swift
   deinit {
       // 先移除视图层级
       flutterVC.willMove(toParent: nil)
       flutterVC.view.removeFromSuperview()
       flutterVC.removeFromParent()
       
       // ✅ 延迟销毁，让 Flutter Widget 有时间完成清理
       DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) { [weak flutterVC] in
           flutterVC?.engine?.destroyContext()
       }
       
       // ✅ 清理 channel 避免内存泄露
       channel?.setMethodCallHandler(nil)
       channel = nil
   }
   ```

### 问题 2：NativeUseFlutterPage.dart 的 useEffect 清理函数不执行

#### 🔴 原始问题：

1. **订阅没有取消**
   ```dart
   return () {
       print("清理函数");
       // sub.cancel();  // ❌ 被注释了
   };
   ```

2. **重复创建 EventChannel**
   - EventMessageChannelManager 内部创建了一个
   - useEffect 中又创建了一个
   - 造成混乱

3. **生命周期异常**
   - iOS 强制销毁 engine 导致 Widget dispose 不执行

#### ✅ 修复方案：

```dart
useEffect(() {
  print("🎯 NativeUseFlutterPage useEffect 开始执行");
  
  // 创建 EventChannel 并监听
  final eventMessageChannel = EventChannel("FlutterNativeEventChannelBridge");
  final subscription = eventMessageChannel.receiveBroadcastStream().listen(
    (event) {
      print("📥 EventChannel 收到事件: $event");
    },
    onError: (error) {
      print("❌ EventChannel 错误: $error");
    },
    onDone: () {
      print("✅ EventChannel 完成");
    },
  );
  
  // ✅ 清理函数：正确取消订阅
  return () {
    print("🧹 NativeUseFlutterPage 清理函数执行 - 取消事件监听");
    subscription.cancel();
  };
}, []);
```

## ⚠️ 关键要点

### 1. Swift 闭包中使用 `[weak self]`
```swift
Timer.scheduledTimer { [weak self] _ in
    guard let self = self else { return }
    // 使用 self
}
```

### 2. Flutter useEffect 清理函数
```dart
useEffect(() {
  final subscription = stream.listen(...);
  return () => subscription.cancel();  // ✅ 必须取消订阅
}, []);
```

### 3. FlutterEngine 销毁顺序
```swift
// 1. 先移除视图
flutterVC.willMove(toParent: nil)
flutterVC.view.removeFromSuperview()
flutterVC.removeFromParent()

// 2. 延迟销毁 context（给 Flutter 时间清理）
DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) {
    flutterVC?.engine?.destroyContext()
}

// 3. 清理 channel
channel?.setMethodCallHandler(nil)
```
